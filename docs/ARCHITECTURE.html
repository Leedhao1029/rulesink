📂 domains/it/orchestrator.py — build_new_l3() 主循环（416行公开）
def _build_new_l3_from_task(self, task: str, gap_info: dict) -> dict:
    # Step 1: 分析需求类型 → 分类为 NEW_SKILL / BROKEN_SCRIPT / ARCHITECTURAL
    capability_type = self._classify_capability_type(task)

    # Step 2: 生成脚本框架（L3Base继承 + SKILL_NAME + 标准方法）
    script_template = self._generate_script_template(capability_type)

    # Step 3: 填充核心逻辑（基于gap分析结果）
    filled = self._fill_script_logic(script_template, gap_info)

    # Step 4: 写入临时目录 + 语法验证（py_compile）
    written_path = self._write_and_verify(filled)

    # Step 5: 熔断门禁检查（SOUL约束层验证）
    safety_result = self._safety_gate_check(written_path)
    if not safety_result["passed"]:
        return {"status": "BLOCKED", "reason": safety_result["reason"]}

    return {"status": "READY", "path": written_path}

class L3Base:
    SCRIPT_NAME = "l3-base"
    VERSION = "1.0"
    DEFAULT_STATUS = "NEEDS_WORK"

    def standard_header(self, quality_status: str = None) -> dict:
        return {
            "script_name": self.SCRIPT_NAME,
            "version": self.VERSION,
            "timestamp": datetime.now().isoformat(),
            "quality_status": quality_status or self.DEFAULT_STATUS,
            "handoff": {
                "ready_for": [],      # 这份输出能给哪些下游脚本用
                "needs_from": [],     # 执行这份任务还需要什么输入

            },

    def analyze(self, task: str) -> dict:         # 标准入口

def diagnose(analysis_result: Dict) -> Dict[str, Any]:
    # Stage 2 DIAGNOSE: 将ANALYZE阶段收集的问题分类
    issues = []
    for script in analysis_result.get("health_metrics", {}).get("scripts_with_issues", []):
        severity   = _classify_severity(script)      # CRITICAL / WARNING / MINOR
        category   = _classify_issue(script)           # MISSING_SKILL / WEAK_SKILL / ...
        recommendation = _get_recommendation(category) # 修复建议文本
        issues.append({
            "item": script, "severity": severity,
            "category": category, "recommendation": recommendation,
        })
    return {
        "stage": "diagnose", "issues": issues,
        "critical_count": sum(1 for i in issues if i["severity"] == "CRITICAL"),
        "warning_count": sum(1 for i in issues if i["severity"] == "WARNING"),
    }

Constraint Types:
  • Role override    — advisor/L3 attempts to direct L2 or L1
  • Decision rights  — advisor output claims decision authority
  • Security red     — unauthorized external API calls
  • Language norm    — prohibited AI jargon / filler words
  • Data integrity   — output missing verification evidence

import concurrent.futures

def convene_council(question: str) -> CouncilReport:
    advisors = [
        StrategicAdvisor(),   # 商业模式 / 市场格局 / 战略一致性
        TechnicalAdvisor(),   # 技术可行性 / 架构风险 / 集成复杂度
        RiskAdvisor(),        # 合规义务 / 安全面 / 尾风险缓释
        EvolutionAdvisor(),   # 模式提炼 / Skill缺口 / 流程改进
    ]

    with concurrent.futures.ThreadPoolExecutor(max_workers=4) as executor:
        futures = {executor.submit(a.analyze, question): a for a in advisors}
        results = {futurs.result() for futu in concurrent.futures.as_completed(futures)}

    synthesis = synthesize(results)    # 四份独立分析 → 综合报告
    audit = it_security_audit(synthesis)  # IT部门审计门禁
    return CouncilReport(advice=synthesis, audit_result=audit)

5步贡献工作流

Step 1  BUILD    → Domain builds new capability using own self-evolution module
Step 2  SELF-TEST → Passes through domain check mode
Step 3  PROPOSE  → Generates contribution proposal (impact analysis)
Step 4  APPROVE  → Tiered: domain-L2 / cross-L2+L1 / all-L2+L1
Step 5  REGISTER  → Add to L3 script library + skill catalog
Step 6  ARCHIVE  → Record in autobiography for traceability
